const http = require('http'); const https = require('https'); const url = require('url'); // Render sets the environment PORT dynamically. Default to 8080 if running locally. const PORT = process.env.PORT || 8080; const server = http.createServer((req, res) => { // Enable Global CORS handling headers so your frontend site can talk to the proxy res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); // Handle preflight browser check requests instantly if (req.method === 'OPTIONS') { res.writeHead(200); res.end(); return; } // Parse out the target destination URL from the query parameter const queryObject = url.parse(req.url, true).query; const targetUrl = queryObject.url; if (!targetUrl) { res.writeHead(400, { 'Content-Type': 'text/plain' }); res.end('Error: Missing target URL parameter (?url=https://...)'); return; } try { const parsedTarget = url.parse(targetUrl); const transport = parsedTarget.protocol === 'https:' ? https : http; const options = { hostname: parsedTarget.hostname, path: parsedTarget.path, method: req.method, headers: { // Emulate a standard browser request setup profile 'User-Agent': req.headers['user-agent'] || 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' } }; // Initialize connection pipeline to the destination game server or website const proxyReq = transport.request(options, (proxyRes) => { const safeHeaders = { ...proxyRes.headers }; // SECURITY BUSTER: Erase the specific headers that force standard browsers to block iframes delete safeHeaders['x-frame-options']; delete safeHeaders['content-security-policy']; // Enforce open sharing origins so the content loads within your client view safeHeaders['Access-Control-Allow-Origin'] = '*'; // Send headers back to your browser window frame layout res.writeHead(proxyRes.statusCode, safeHeaders); // Directly stream data packets from the target website back to the user screen proxyRes.pipe(res); }); proxyReq.on('error', (err) => { res.writeHead(500, { 'Content-Type': 'text/plain' }); res.end(`Proxy connection failed to resolve target host stream: ${err.message}`); }); // Pass any inbound user payload requests down the connection pipe req.pipe(proxyReq); } catch (error) { res.writeHead(500, { 'Content-Type': 'text/plain' }); res.end('Invalid URL payload processed by proxy server parsing rules.'); } }); // Run engine initialization routine listener server.listen(PORT, () => { console.log(`Proxy core actively running on processing port ${PORT}`); });
